1"""Reader for the dbt Information Schema (dbt 2.0+). 

2 

3dbt writes the Information Schema as Parquet files under 

4``<target>/info_schema/v1/`` when a command runs with ``--generate-info-schema``. 

5dbt documents this directory as a contracted, versioned interface, so dbt-bouncer 

6reads it rather than the undocumented ``<target>/private/`` index. See 

7https://docs.getdbt.com/reference/info-schema. 

8 

9``duckdb`` is imported lazily: only runs that configure ``info_schema_checks`` 

10pay for the import. 

11""" 

12

36# The only Information Schema version dbt publishes so far. The directory name 

37# (`v1`) and `dbt.project.schema_version` both carry it. 

38INFO_SCHEMA_VERSION = 1 

39 

40# `copy` and `mod` edges carry a parent column's value into the child column. 

41# `scan` edges only mean the parent column was read (e.g. in a join or filter). 

42VALUE_LINEAGE_KINDS = frozenset({"copy", "mod"}) 

43 

44# Parquet files are named `<schema>.<table>.parquet`, e.g. `dbt.models.parquet`. 

45_TABLE_FILE_PATTERN = re.compile(r"^([a-z_][a-z0-9_]*)\.([a-z_][a-z0-9_]*)$")

58def require_single_select(sql: str) -> None: 

59 """Reject SQL that is not exactly one SELECT statement. 

60 

61 The Information Schema database also has external access disabled; this 

62 stops statements such as `COPY` or `ATTACH` before they run. 

63 

64 Raises: 

65 ValueError: If `sql` does not parse, or is not a single SELECT. 

66 

67 """ 

68 import duckdb 

69 

70 try: 

71 statements = duckdb.extract_statements(sql) 

72 except duckdb.Error as e: 

73 raise ValueError(f"`sql` is not valid SQL: {e}") from e 

74 if len(statements) != 1 or statements[0].type != duckdb.StatementType.SELECT: 

75 raise ValueError("`sql` must be exactly one SELECT statement.")

89class InfoSchema: 

90 """Tables from the dbt Information Schema, plus lookups the checks share. 

91 

92 Build one with :meth:`from_directory` (production) or :meth:`from_rows` 

93 (unit tests). Every table is a list of row dicts keyed by column name. 

94 Column names are matched case-insensitively, because warehouses such as 

95 Snowflake upper-case them in some tables but not others. 

96 """ 

97 

98 def __init__( 

99 self, 

100 tables: dict[str, list[dict[str, Any]]] | None = None, 

101 directory: Path | None = None, 

102 ) -> None: 

103 """Store pre-loaded tables or the directory to load them from lazily.""" 

104 self._tables: dict[str, list[dict[str, Any]]] = dict(tables or {}) 

105 self.directory = directory 

106 

107 @classmethod 

108 def from_directory(cls, directory: Path) -> InfoSchema: 

109 """Open an Information Schema directory and confirm its version. 

110 

111 Args: 

112 directory: The ``info_schema/v1`` directory. 

113 

114 Returns: 

115 InfoSchema: A reader backed by ``directory``. 

116 

117 Raises: 

118 DbtBouncerArtifactError: If the directory is missing or reports an 

119 unsupported ``schema_version``. 

120 

121 """ 

122 if not (directory / "dbt.project.parquet").exists(): 

123 raise DbtBouncerArtifactError( 

124 f"No dbt Information Schema found at `{directory}`. It requires dbt 2.0 or later: " 

125 "run `dbt build --generate-info-schema` (or `dbt compile --generate-info-schema`)." 

126 ) 

127 info_schema = cls(directory=directory) 

128 project_rows = info_schema.table("dbt.project") 

129 versions = {row.get("schema_version") for row in project_rows} 

130 if versions != {INFO_SCHEMA_VERSION}: 

131 raise DbtBouncerArtifactError( 

132 f"`{directory}` has Information Schema version {sorted(versions, key=str)}, " 

133 f"dbt-bouncer supports version {INFO_SCHEMA_VERSION} only." 

134 ) 

135 return info_schema 

136 

137 @classmethod 

138 def from_rows(cls, **tables: list[dict[str, Any]]) -> InfoSchema: 

139 """Build an in-memory Information Schema, for unit tests. 

140 

141 Pass each table as a keyword argument named after the table without its 

142 ``dbt.`` schema, e.g. ``from_rows(column_lineage=[...], models=[...])``. 

143 

144 Returns: 

145 InfoSchema: A reader over the given rows. 

146 

147 """ 

148 return cls(tables={f"dbt.{name}": rows for name, rows in tables.items()})

150 @cached_property 

151 def table_files(self) -> dict[str, Path]: 

152 """The Parquet file of each table, keyed by ``<schema>.<table>``. 

153 

154 Files whose names are not a valid ``<schema>.<table>`` are skipped. 

155 

156 Returns: 

157 dict[str, Path]: The table files; empty for a reader built from rows. 

158 

159 """ 

160 if self.directory is None: 

161 return {} 

162 return { 

163 parquet_file.stem: parquet_file 

164 for parquet_file in sorted(self.directory.glob("*.parquet")) 

165 if _TABLE_FILE_PATTERN.match(parquet_file.stem) 

166 }

168 @cached_property 

169 def connection(self) -> duckdb.DuckDBPyConnection: 

170 """An in-memory DuckDB database holding every Information Schema table. 

171 

172 Each ``<schema>.<table>.parquet`` file becomes the table 

173 ``<schema>.<table>``. After loading, external access is disabled and the 

174 configuration locked, so SQL from a config file cannot read or write files. 

175 

176 Returns: 

177 duckdb.DuckDBPyConnection: The connection. 

178 

179 Raises: 

180 DbtBouncerArtifactError: If the reader has no backing directory. 

181 

182 """ 

183 if self.directory is None: 

184 raise DbtBouncerArtifactError( 

185 "This Information Schema was built from in-memory rows and has no database." 

186 ) 

187 import duckdb 

188 

189 con = duckdb.connect(":memory:") 

190 for name, parquet_file in self.table_files.items(): 

191 schema, table = name.split(".") 

192 con.execute(f'CREATE SCHEMA IF NOT EXISTS "{schema}"') 

193 # Identifiers come from the allow-list regex above; the path is a bound parameter. 

194 con.execute( 

195 f'CREATE TABLE "{schema}"."{table}" AS SELECT * FROM read_parquet(?)', # ruff: ignore[hardcoded-sql-expression] # nosec B608 

196 [parquet_file.as_posix()], 

197 ) 

198 con.execute("SET enable_external_access = false") 

199 con.execute("SET lock_configuration = true") 

200 return con

202 def query(self, sql: str) -> tuple[list[str], list[tuple[str | None, ...]]]: 

203 """Run a read-only SELECT against the Information Schema. 

204 

205 Every value comes back as text, so results print as-is and time zone 

206 aware timestamps need no extra Python packages. 

207 

208 ``sql`` is checked to be a single SELECT before it runs (``require_single_select`` 

209 raises ``ValueError`` otherwise), so this method is safe to call from any path. ``check_info_schema_query`` also validates its 

210 ``sql`` at config load, to fail before the run starts. External access is 

211 disabled on the connection, so no SQL can read or write files. 

212 

213 Args: 

214 sql: A single SELECT statement. 

215 

216 Returns: 

217 tuple[list[str], list[tuple[str | None, ...]]]: Column names and rows. 

218 

219 """ 

220 require_single_select(sql) 

221 cursor = self.connection.execute( 

222 f"SELECT CAST(COLUMNS(*) AS VARCHAR) FROM ({sql.strip().rstrip(';')})" # ruff: ignore[hardcoded-sql-expression] # nosec B608 

223 ) 

224 columns = [d[0] for d in cursor.description or []] 

225 return columns, cursor.fetchall()

227 def table(self, name: str) -> list[dict[str, Any]]: 

228 """Return every row of a table, e.g. ``table("dbt.models")``. 

229 

230 A table that the directory does not contain returns no rows. 

231 

232 Returns: 

233 list[dict[str, Any]]: The rows. 

234 

235 """ 

236 if name not in self._tables: 

237 rows: list[dict[str, Any]] = [] 

238 if self.directory is not None: 

239 schema, table = name.split(".", 1) 

240 exists = self.connection.execute( 

241 "SELECT 1 FROM information_schema.tables WHERE table_schema = ? AND table_name = ?", 

242 [schema, table], 

243 ).fetchone() 

244 if exists: 

245 rows = self._read_table(schema, table) 

246 self._tables[name] = rows 

247 return self._tables[name] 

248 

249 def _read_table(self, schema: str, table: str) -> list[dict[str, Any]]: 

250 """Read a table into row dicts, keeping list columns (e.g. ``grain``) as lists. 

251 

252 Time zone aware timestamps are cast to text: converting them to Python 

253 objects requires the ``pytz`` package. 

254 

255 Returns: 

256 list[dict[str, Any]]: The rows. 

257 

258 """ 

259 column_types = self.connection.execute( 

260 "SELECT column_name, data_type FROM information_schema.columns " 

261 "WHERE table_schema = ? AND table_name = ? ORDER BY ordinal_position", 

262 [schema, table], 

263 ).fetchall() 

264 select_list = ", ".join( 

265 f'CAST("{name}" AS VARCHAR) AS "{name}"' 

266 if data_type == "TIMESTAMP WITH TIME ZONE" 

267 else f'"{name}"' 

268 for name, data_type in column_types 

269 ) 

270 cursor = self.connection.execute( 

271 f'SELECT {select_list} FROM "{schema}"."{table}"' # ruff: ignore[hardcoded-sql-expression] # nosec B608 

272 ) 

273 columns = [d[0] for d in cursor.description or []] 

274 return [dict(zip(columns, row, strict=True)) for row in cursor.fetchall()]

276 @cached_property 

277 def lineage_edges(self) -> list[LineageEdge]: 

278 """Every column-level lineage edge. 

279 

280 Returns: 

281 list[LineageEdge]: The edges. 

282 

283 """ 

284 return [ 

285 LineageEdge( 

286 parent_node_unique_id=row["parent_node_unique_id"], 

287 parent_column_name=row["parent_column_name"], 

288 child_node_unique_id=row["child_node_unique_id"], 

289 child_column_name=row["child_column_name"], 

290 evolution=row["evolution"], 

291 ) 

292 for row in self.table("dbt.column_lineage") 

293 ]

295 @cached_property 

296 def lineage_by_child(self) -> dict[str, dict[str, list[LineageEdge]]]: 

297 """Incoming lineage edges keyed by child node unique_id, then lower-cased child column. 

298 

299 Returns: 

300 dict[str, dict[str, list[LineageEdge]]]: The edges per child column. 

301 

302 """ 

303 by_child: dict[str, dict[str, list[LineageEdge]]] = {} 

304 for edge in self.lineage_edges: 

305 by_child.setdefault(edge.child_node_unique_id, {}).setdefault( 

306 edge.child_column_name.casefold(), [] 

307 ).append(edge) 

308 return by_child 

309 

310 @cached_property 

311 def lineage_parent_columns(self) -> dict[str, set[str]]: 

312 """Lower-cased column names with at least one downstream edge, per node. 

313 

314 Returns: 

315 dict[str, set[str]]: Node unique_id to its consumed column names. 

316 

317 """ 

318 consumed: dict[str, set[str]] = {} 

319 for edge in self.lineage_edges: 

320 consumed.setdefault(edge.parent_node_unique_id, set()).add( 

321 edge.parent_column_name.casefold() 

322 ) 

323 return consumed

335 @cached_property 

336 def node_columns(self) -> dict[str, dict[str, dict[str, Any]]]: 

337 """Rows of ``dbt.node_columns`` keyed by node unique_id, then lower-cased column name. 

338 

339 Returns: 

340 dict[str, dict[str, dict[str, Any]]]: The column rows per node. 

341 

342 """ 

343 by_node: dict[str, dict[str, dict[str, Any]]] = {} 

344 for row in self.table("dbt.node_columns"): 

345 by_node.setdefault(row["node_unique_id"], {})[ 

346 row["column_name"].casefold() 

347 ] = row 

348 return by_node
