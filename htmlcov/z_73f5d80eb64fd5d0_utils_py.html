14def category_key(check_class: type["BaseCheck"]) -> str: 

15 """Return the display category name segment from the module path. 

16 

17 Args: 

18 check_class: The check class whose module path is inspected. 

19 

20 Returns: 

21 str: The category name segment. 

22 

23 """ 

24 # e.g. "dbt_bouncer.checks.manifest.check_models" -> "manifest"

62def get_check_params(check_class: type["BaseCheck"]) -> dict[str, str]: 

63 """Return configurable parameter names and their type annotations. 

64 

65 Args: 

66 check_class: The check class whose model fields are inspected. 

67 

68 Returns: 

69 dict[str, str]: Mapping of field name to type string. 

70 

71 """ 

72 params: dict[str, str] = {} 

73 for field_name, field_info in check_class.model_fields.items(): 

74 if field_name in base_fields: 

75 continue 

76 annotation = field_info.annotation 

77 type_str = getattr(annotation, "__name__", None) or str(annotation) 

78 params[field_name] = type_str 

79 return params

91def build_checks_payload( 

92 checks: list[type["BaseCheck"]], category_labels: dict[str, str] 

93) -> dict[str, list[CheckDict]]: 

94 """Build the checks payload grouped by category. 

95 

96 Args: 

97 checks: List of check classes. 

98 category_labels: Mapping from category to category label. 

99 

100 Returns: 

101 The checks payload grouped by category. 

102 

103 """ 

104 result: dict[str, list[CheckDict]] = {} 

105 for category, group in itertools.groupby(checks, key=category_key): 

106 label = category_labels.get(category, category) 

107 result[label] = [] 

108 for check_class in group: 

109 docstring = (check_class.__doc__ or "").strip() 

110 description = docstring.splitlines()[0] if docstring else "" 

111 result[label].append( 

112 { 

113 "code": getattr(check_class, "code", None), 

114 "description": description, 

115 "name": check_class.__name__, 

116 "parameters": get_check_params(check_class), 

117 } 

118 ) 

119 return result

122def print_text_checks(checks_payload: dict[str, list[CheckDict]]) -> None: 

123 """Print checks formatted in Rich tables grouped by category. 

124 

125 Args: 

126 checks_payload: Checks grouped by category label, as returned by 

127 :func:`build_checks_payload`. 

128 

129 """ 

130 console = Console() 

131 for label, checks in checks_payload.items(): 

132 table = Table( 

133 title=f"[bold cyan]{label}[/bold cyan]", 

134 title_justify="left", 

135 box=box.ROUNDED, 

136 border_style="cyan", 

137 show_header=True, 

138 header_style="bold cyan", 

139 ) 

140 table.add_column("Code", style="bold cyan", justify="center", no_wrap=True) 

141 table.add_column("Check Name", style="bold white", justify="left", no_wrap=True) 

142 table.add_column("Description", justify="left") 

143 table.add_column("Parameters", justify="left", style="yellow") 

144 

145 for check in checks: 

146 code = check.get("code") or "-" 

147 user_params = {k: v for k, v in check["parameters"].items() if k != "name"} 

148 params_text = ( 

149 "\n".join( 

150 f"{name}: {type_str}" for name, type_str in user_params.items() 

151 ) 

152 if user_params 

153 else "(none)" 

154 ) 

155 table.add_row( 

156 code, 

157 check["name"], 

158 check["description"], 

159 params_text, 

160 ) 

161 

162 console.print(table) 

163 console.print()
