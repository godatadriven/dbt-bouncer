1"""Create or update one dbt-bouncer comment on a GitHub pull request. 

2 

3Each run edits the same comment instead of adding a new one. A hidden HTML 

4marker at the start of the body identifies the comment. The marker includes a 

5key (the config file), so two dbt-bouncer jobs on one PR keep separate comments. 

6 

7The GitHub API is called with `urllib` from the standard library, so the 

8feature adds no dependency. Any failure to post is logged as a warning and never 

9changes the exit code: the check result is what gates the PR, not the comment. 

10""" 

11

50def build_marker(comment_key: str) -> str: 

51 """Build the hidden marker that identifies the dbt-bouncer comment. 

52 

53 Returns: 

54 str: An HTML comment, invisible in the rendered PR comment. 

55 

56 """ 

57 # `--` cannot appear inside an HTML comment, so it would end the marker early. 

58 return f"<!-- dbt-bouncer-pr-comment: {re.sub(r'-{2,}', '-', comment_key)} -->"

61def _passed_body(pull_request: _PullRequest) -> str: 

62 """Build the comment body for a run where every check passed. 

63 

64 Returns: 

65 str: The markdown comment body. 

66 

67 """ 

68 commit = f" as of commit `{pull_request.head_sha}`" if pull_request.head_sha else "" 

69 source = ( 

70 f" Sent from this [GitHub Action workflow run]({pull_request.run_url})." 

71 if pull_request.run_url 

72 else "" 

73 ) 

74 return f"## **Passed `dbt-bouncer`** checks\n\nAll checks passed{commit}.{source}"

77def _read_pull_request() -> _PullRequest | None: 

78 """Read the pull request details from the GitHub Actions environment. 

79 

80 Returns: 

81 _PullRequest | None: The pull request, or `None` if the run is not for a 

82 pull request or a required variable is missing. The reason is logged. 

83 

84 """ 

85 token = os.environ.get("GITHUB_TOKEN") or os.environ.get("GH_TOKEN") 

86 repository = os.environ.get("GITHUB_REPOSITORY") 

87 event_path = os.environ.get("GITHUB_EVENT_PATH") 

88 if not token or not repository or not event_path: 

89 logging.warning( 

90 "`--github-pr-comment` is set, but GITHUB_TOKEN (or GH_TOKEN), GITHUB_REPOSITORY or GITHUB_EVENT_PATH is not. No PR comment is sent. Pass `GITHUB_TOKEN: ${{ github.token }}` in the step's `env:`." 

91 ) 

92 return None 

93 

94 try: 

95 event = orjson.loads(Path(event_path).read_bytes()) 

96 except (OSError, orjson.JSONDecodeError) as e: 

97 logging.warning( 

98 f"Could not read the GitHub event file `{event_path}`, so no PR comment is sent: {e}" 

99 ) 

100 return None 

101 

102 pull_request = event.get("pull_request") if isinstance(event, dict) else None 

103 if not isinstance(pull_request, dict) or "number" not in pull_request: 

104 logging.warning( 

105 "`--github-pr-comment` is set, but this workflow run is not for a pull request. No PR comment is sent." 

106 ) 

107 return None 

108 

109 run_id = os.environ.get("GITHUB_RUN_ID") 

110 server_url = os.environ.get("GITHUB_SERVER_URL", "https://github.com").rstrip("/") 

111 return _PullRequest( 

112 api_url=os.environ.get("GITHUB_API_URL", _DEFAULT_API_URL).rstrip("/"), 

113 head_sha=str((pull_request.get("head") or {}).get("sha") or ""), 

114 number=int(pull_request["number"]), 

115 repository=repository, 

116 run_url=f"{server_url}/{repository}/actions/runs/{run_id}" if run_id else None, 

117 token=token, 

118 )

121def _github_request( 

122 method: str, url: str, token: str, payload: Mapping[str, Any] | None = None 

123) -> Any: 

124 """Send one request to the GitHub REST API. 

125 

126 Returns: 

127 Any: The decoded JSON response. 

128 

129 Raises: 

130 ValueError: If the URL is not HTTPS. 

131 

132 """ 

133 # GITHUB_API_URL comes from the environment, so refuse `file:` and other schemes. 

134 if not url.startswith("https://"): 

135 raise ValueError(f"Refusing to call a non-HTTPS GitHub API URL: `{url}`.") 

136 request = urllib.request.Request( # ruff: ignore[suspicious-url-open-usage] - scheme checked above 

137 url, 

138 data=orjson.dumps(payload) if payload is not None else None, 

139 headers={ 

140 "Accept": "application/vnd.github+json", 

141 "Authorization": f"Bearer {token}", 

142 "Content-Type": "application/json", 

143 "X-GitHub-Api-Version": "2022-11-28", 

144 }, 

145 method=method, 

146 ) 

147 with urllib.request.urlopen(request, timeout=_REQUEST_TIMEOUT_SECONDS) as response: # ruff: ignore[suspicious-url-open-usage] # nosec B310 

148 return orjson.loads(response.read() or b"null")

151def _find_comment_id(pull_request: _PullRequest, marker: str) -> int | None: 

152 """Find the first comment on the pull request that starts with `marker`. 

153 

154 Each run edits that one comment, so normally only one has the marker. Paging 

155 stops at the first match. 

156 

157 Returns: 

158 int | None: The comment ID, or `None` if no comment has the marker. 

159 

160 """ 

161 page = 1 

162 while True: 

163 comments = _github_request( 

164 "GET", 

165 f"{pull_request.api_url}/repos/{pull_request.repository}/issues/{pull_request.number}/comments?per_page={_COMMENTS_PER_PAGE}&page={page}", 

166 pull_request.token, 

167 ) 

168 for comment in comments: 

169 # `startswith`, not `in`: a reply that quotes the comment starts with `>`. 

170 if str(comment.get("body") or "").startswith(marker): 

171 return int(comment["id"]) 

172 if len(comments) < _COMMENTS_PER_PAGE: 

173 return None 

174 page += 1

177def sync_github_pr_comment( 

178 results: Sequence[Mapping[str, Any]], 

179 *, 

180 comment_key: str, 

181 show_all_failures: bool = False, 

182) -> None: 

183 """Create or update the dbt-bouncer comment on the current pull request. 

184 

185 - If a check failed or warned, the comment lists those checks. An earlier 

186 comment from the same config is edited, else a new comment is created. 

187 - If every check passed, an earlier comment is edited to say so. If there is 

188 no earlier comment, nothing is posted, so clean PRs stay free of comments. 

189 

190 Errors are logged as warnings and never raised. 

191 

192 Args: 

193 results: The check results returned by the runner. 

194 comment_key: Identifies the comment, so separate configs keep separate comments. 

195 show_all_failures: List every failure, not only the first 25. 

196 

197 """ 

198 pull_request = _read_pull_request() 

199 if pull_request is None: 

200 return 

201 

202 marker = build_marker(comment_key) 

203 failed_checks = [ 

204 [ 

205 str(r["check_run_id"]), 

206 str(r.get("file_path") or ""), 

207 str(r.get("severity", "")).lower(), 

208 str(r.get("failure_message", "")), 

209 ] 

210 for r in results 

211 if r["outcome"] != CheckOutcome.SUCCESS 

212 ] 

213 

214 try: 

215 comment_id = _find_comment_id(pull_request, marker) 

216 if failed_checks: 

217 body = build_github_comment_body(failed_checks, show_all_failures) 

218 elif comment_id is not None: 

219 body = _passed_body(pull_request) 

220 else: 

221 logging.info("All checks passed and no earlier PR comment exists.") 

222 return 

223 

224 payload = {"body": f"{marker}\n{body}"} 

225 if comment_id is None: 

226 _github_request( 

227 "POST", 

228 f"{pull_request.api_url}/repos/{pull_request.repository}/issues/{pull_request.number}/comments", 

229 pull_request.token, 

230 payload, 

231 ) 

232 logging.info(f"Created a comment on PR #{pull_request.number}.") 

233 else: 

234 _github_request( 

235 "PATCH", 

236 f"{pull_request.api_url}/repos/{pull_request.repository}/issues/comments/{comment_id}", 

237 pull_request.token, 

238 payload, 

239 ) 

240 logging.info(f"Updated comment {comment_id} on PR #{pull_request.number}.") 

241 except urllib.error.HTTPError as e: 

242 hint = ( 

243 " Give the job `permissions: pull-requests: write`." 

244 if e.code in {401, 403, 404} 

245 else "" 

246 ) 

247 logging.warning( 

248 f"Could not send the PR comment: GitHub API returned HTTP {e.code}.{hint}" 

249 ) 

250 except (OSError, ValueError) as e: 

251 logging.warning(f"Could not send the PR comment: {e}")
