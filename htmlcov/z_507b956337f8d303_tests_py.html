1"""Checks related to column tests.""" 

2 

3import re 

4 

5from dbt_bouncer.check_framework.decorator import check, fail 

6from dbt_bouncer.types import RegexPattern 

7from dbt_bouncer.utils import compile_pattern 

8 

9 

10@check(code="CA011") 

11def check_column_has_specified_test( 

12 catalog_node, 

13 ctx, 

14 *, 

15 column_name_pattern: RegexPattern, 

16 test_name: str, 

17 case_sensitive: bool = True, 

18): 

19 """Columns that match the specified regexp pattern must have a specified test. 

20 

21 !!! info "Rationale" 

22 

23 Naming conventions communicate expectations: a column named `is_active` implies it is boolean and never null; a column ending in `_id` implies it is a valid foreign key. Without enforcement, these implicit contracts go untested, and referential integrity issues or null values can silently corrupt downstream aggregations. This check bridges naming conventions and data quality by automatically requiring specific tests on columns that match a pattern, eliminating the manual overhead of reviewing every column individually. 

24 

25 Parameters: 

26 case_sensitive (bool): Whether the column names are case sensitive or not. Necessary for adapters like `dbt-snowflake` where the column in `catalog.json` is uppercase but the column in `manifest.json` can be lowercase. Defaults to `false` for `dbt-snowflake`, otherwise `true`. 

27 column_name_pattern (str): Regex pattern to match the column name. 

28 test_name (str): Name of the test to check for. 

29 

30 Receives: 

31 catalog_node (CatalogNodeEntry): The CatalogNodeEntry object to check. 

32 manifest_obj (ManifestObject): The ManifestObject object parsed from `manifest.json`. 

33 tests (list[TestNode]): List of TestNode objects parsed from `manifest.json`. 

34 

35 Other Parameters: 

36 description (str | None): Description of what the check does and why it is implemented. 

37 exclude (str | list[str] | None): Regex pattern(s) to match the model path. Model paths that match any pattern will not be checked. 

38 include (str | list[str] | None): Regex pattern(s) to match the model path. Only model paths that match any pattern will be checked. 

39 severity (Literal["error", "warn"] | None): Severity level of the check. Default: `error`. 

40 

41 Example(s): 

42 ```yaml 

43 catalog_checks: 

44 - name: check_column_has_specified_test 

45 column_name_pattern: ^is_.* 

46 test_name: not_null 

47 ``` 

48 

49 """ 

50 if ctx.manifest_obj.manifest.metadata.adapter_type in ["snowflake"]: 

51 case_sensitive = False 

52 

53 # When matching is case-insensitive the catalog name is upper-cased (e.g. 

54 # Snowflake) while patterns are conventionally written lowercase, so the regex 

55 # must ignore case too — otherwise the column silently falls out of scope. 

56 pattern_flags = 0 if case_sensitive else re.IGNORECASE 

57 compiled_column_name_pattern = compile_pattern( 

58 column_name_pattern.strip(), pattern_flags 

59 ) 

60 columns_to_check = [ 

61 v.name 

62 for _, v in catalog_node.columns.items() 

63 if compiled_column_name_pattern.match(str(v.name)) is not None 

64 ] 

65 tested_columns = set() 

66 for t in ctx.tests_by_attached_node.get(catalog_node.unique_id, []): 

67 test_metadata = getattr(t, "test_metadata", None) 

68 if test_metadata and getattr(test_metadata, "name", None) == test_name: 

69 column_name = getattr(t, "column_name", "") 

70 # A column-level test exposes the documented (YAML) column name; model-level 

71 # tests have no `column_name`, so skip them rather than adding an empty entry. 

72 if column_name: 

73 tested_columns.add(column_name) 

74 

75 if case_sensitive: 

76 non_complying_columns = [c for c in columns_to_check if c not in tested_columns] 

77 else: 

78 # On case-folding adapters (e.g. Snowflake) the catalog column is uppercase 

79 # while the test's column name mirrors the lowercase YAML, so compare lowered. 

80 tested_columns_lower = {c.lower() for c in tested_columns} 

81 non_complying_columns = [ 

82 c for c in columns_to_check if c.lower() not in tested_columns_lower 

83 ] 

84 

85 if non_complying_columns: 

86 fail( 

87 f"`{str(catalog_node.unique_id).split('.')[-1]}` has columns that should have a `{test_name}` test: {non_complying_columns}" 

88 )
