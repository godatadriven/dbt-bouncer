21class Executor: 

22 """Orchestrates check execution with progress tracking.""" 

23 

24 # Resolved once per ``run()``. Formatting a per-check debug message costs an 

25 # f-string and a logging call for every check -- tens of thousands on a large 

26 # project -- all of it discarded when DEBUG is off, which is the default. 

27 _debug_enabled: bool = False 

28 

29 def _execute_check(self, check: CheckToRun) -> CheckToRun: 

30 """Execute a single check and return the result. 

31 

32 Returns: 

33 CheckToRun: The check dict with outcome and optional failure_message set. 

34 

35 """ 

36 if self._debug_enabled: 

37 logging.debug(f"Running {check['check_run_id']}...") 

38 # Bind this run's resource onto the shared check instance immediately 

39 # before executing. Assembly stores the resource alongside the check 

40 # rather than pre-copying an instance per resource; sequential execution 

41 # makes reusing one instance across its resources safe. Context-only 

42 # checks have no resource and are executed as-is. 

43 resource = check.get("resource") 

44 if resource is not None: 

45 check["check"].set_resource(resource, check["iterate_value"]) 

46 try: 

47 check["check"].execute() 

48 check["outcome"] = CheckOutcome.SUCCESS 

49 except DbtBouncerFailedCheckError as e: 

50 failure_message = e.message 

51 if check["check"].description: 

52 failure_message = f"{check['check'].description} - {failure_message}" 

53 

54 if self._debug_enabled: 

55 logging.debug( 

56 f"Check {check['check_run_id']} failed: {failure_message}" 

57 ) 

58 check["outcome"] = CheckOutcome.FAILED 

59 check["failure_message"] = failure_message 

60 except Exception as e: 

61 failure_message_full = list( 

62 traceback.TracebackException.from_exception(e).format(), 

63 ) 

64 failure_message = failure_message_full[-1].strip() 

65 if self._debug_enabled: 

66 logging.debug( 

67 f"Check {check['check_run_id']} raised unexpected error:\n{''.join(failure_message_full)}" 

68 ) 

69 

70 # Fail closed: a check that crashed has not verified anything, so it 

71 # keeps its configured severity (``error`` by default) rather than 

72 # being downgraded to a warning that would let the run pass. 

73 check["outcome"] = CheckOutcome.INTERNAL_ERROR 

74 check["failure_message"] = ( 

75 f"`dbt-bouncer` encountered an error ({failure_message}), run with `-v` to see more details or report an issue at https://github.com/godatadriven/dbt-bouncer/issues." 

76 ) 

77 return check

79 def run(self, checks_to_run: list[CheckToRun]) -> list[dict[str, Any]]: 

80 """Execute all checks sequentially with progress tracking. 

81 

82 Args: 

83 checks_to_run: List of CheckToRun dicts (mutated in place during execution). 

84 

85 Returns: 

86 list[dict]: Result dicts with check_run_id, failure_message, file_path, 

87 outcome, severity, unique_id. 

88 

89 """ 

90 if not checks_to_run: 

91 logging.info("No checks to run.") 

92 return [] 

93 

94 logging.info(f"Assembled {len(checks_to_run)} checks, running...") 

95 

96 self._debug_enabled = logging.getLogger().isEnabledFor(logging.DEBUG) 

97 

98 # Checks are CPU-bound pure-Python work (regex matching, proxy attribute 

99 # access, string ops) that never releases the GIL, so a ThreadPoolExecutor 

100 # could not run them in parallel -- it only added thread-scheduling and 

101 # GIL-contention overhead (and multi-second tail-latency spikes on large 

102 # projects). Executing sequentially is both faster and far more 

103 # predictable; see the benchmark suite (``tests/benchmark``). 

104 console = Console() 

105 with Progress( 

106 TextColumn("[progress.description]{task.description}"), 

107 BarColumn(), 

108 TaskProgressColumn(), 

109 console=console, 

110 ) as progress: 

111 total = len(checks_to_run) 

112 task = progress.add_task("Running checks...", total=total) 

113 # Refresh the bar ~100 times total rather than once per check: at large 

114 # check counts per-check updates add measurable overhead, and rich only 

115 # renders a few times a second anyway. 

116 update_step = max(1, total // 100) 

117 for n, check in enumerate(checks_to_run, 1): 

118 self._execute_check(check) 

119 if n % update_step == 0: 

120 progress.update(task, completed=n) 

121 progress.update(task, completed=total) 

122 

123 return [ 

124 { 

125 "check_run_id": c["check_run_id"], 

126 "failure_message": c.get("failure_message"), 

127 "file_path": c.get("file_path"), 

128 "outcome": c["outcome"], 

129 "severity": c["severity"], 

130 "unique_id": c.get("unique_id"), 

131 } 

132 for c in checks_to_run 

133 ]
